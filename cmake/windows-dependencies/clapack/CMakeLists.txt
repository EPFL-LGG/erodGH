cmake_minimum_required(VERSION 3.20)
project(ErodReferenceLAPACK C)
if(NOT EROD_CLAPACK_SOURCE)
    message(FATAL_ERROR "EROD_CLAPACK_SOURCE is required")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
add_subdirectory("${EROD_CLAPACK_SOURCE}" upstream EXCLUDE_FROM_ALL)
add_custom_target(erod_lapack ALL DEPENDS lapack blas f2c)
install(TARGETS lapack blas f2c ARCHIVE DESTINATION lib)
configure_file(lapack-config.cmake.in lapack-config.cmake @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/lapack-config.cmake" DESTINATION lib/cmake/lapack)
install(FILES "${EROD_CLAPACK_SOURCE}/COPYING" DESTINATION share/licenses/clapack)
