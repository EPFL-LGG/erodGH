cmake_minimum_required(VERSION 3.20)
project(erod_interop_tests LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This standalone loader harness currently targets Windows")
endif()
option(EROD_TEST_ASAN "Instrument the test executable with AddressSanitizer" ON)
add_executable(erod-interop-tests main.cpp)
if(EROD_TEST_ASAN)
    target_compile_options(erod-interop-tests PRIVATE /fsanitize=address /Z7 /Od)
    target_link_options(erod-interop-tests PRIVATE /DEBUG)
endif()
