cmake_minimum_required(VERSION 3.20)
project(erod_interop_tests LANGUAGES CXX)
option(EROD_TEST_ASAN "Instrument the test executable with AddressSanitizer" ON)
add_executable(erod-interop-tests main.cpp)
if(EROD_TEST_ASAN AND MSVC)
    target_compile_options(erod-interop-tests PRIVATE /fsanitize=address /Z7 /Od)
    target_link_options(erod-interop-tests PRIVATE /DEBUG)
endif()

if(EROD_TEST_ASAN AND NOT MSVC)
    target_compile_options(erod-interop-tests PRIVATE -fsanitize=address -g)
    target_link_options(erod-interop-tests PRIVATE -fsanitize=address)
endif()
target_link_libraries(erod-interop-tests PRIVATE ${CMAKE_DL_LIBS})
set(EROD_TEST_BINARY_DIRECTORY "" CACHE PATH "Directory containing the ERod runtime")
if(EROD_TEST_BINARY_DIRECTORY)
    enable_testing()
    foreach(_case error hardening materials buffers solver equilibrium bad-dofs strain lifetime)
        add_test(NAME interop-${_case} COMMAND erod-interop-tests "${EROD_TEST_BINARY_DIRECTORY}" "${_case}")
    endforeach()
endif()
